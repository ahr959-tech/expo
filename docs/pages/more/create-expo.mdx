# Backend FastAPI starter
backend_dir = os.path.join(root, "backend")
os.makedirs(backend_dir, exist_ok=True)
main_py = textwrap.dedent("""\
    from fastapi import FastAPI, HTTPException
    from pydantic import BaseModel
    from typing import List
    app = FastAPI(title='Nabta API (starter)')
   
    class Product(BaseModel):
        id: str
        name: str
        short: str
        description: str
        price: float
   
    @app.get('/health')
    def health():
        return {'status':'ok'}
   
    # sample products - in prod load from DB
    PRODUCTS = []
    try:
        import json, os
        fn = os.path.join(os.path.dirname(__file__), '..','frontend','data','product_catalog.json')
        with open(fn, 'r', encoding='utf-8') as fh:
            PRODUCTS = json.load(fh)
    except Exception:
        PRODUCTS = []
   
    @app.get('/store', response_model=List[Product])
    def list_products():
        return PRODUCTS
   
    @app.get('/store/{product_id}', response_model=Product)
    def get_product(product_id: str):
        for p in PRODUCTS:
            if p['id']==product_id:
                return p
        raise HTTPException(status_code=404, detail='product not found')
    """)

with open(os.path.join(backend_dir, "main.py"), "w", encoding="utf-8") as f:

# DB schema
db_schema = textwrap.dedent("""\
    -- Simple PostgreSQL schema for NABTA starter
    CREATE TABLE users (
      id SERIAL PRIMARY KEY,
      name TEXT,
      email TEXT UNIQUE,
      hashed_password TEXT,
      role TEXT,
      created_at TIMESTAMP DEFAULT now()
    );
    CREATE TABLE systems (
      id SERIAL PRIMARY KEY,
      user_id INTEGER REFERENCES users(id),
      name TEXT,
      location TEXT,
      hardware_info JSONB,
      created_at TIMESTAMP DEFAULT now()
    );
    CREATE TABLE sensor_readings (
      id BIGSERIAL PRIMARY KEY,
      system_id INTEGER REFERENCES systems(id),
      timestamp TIMESTAMP DEFAULT now(),
      ph NUMERIC,
      ec NUMERIC,
      temp_c NUMERIC,
      water_level NUMERIC,
      light_lux NUMERIC
    );
    CREATE TABLE products (
      id TEXT PRIMARY KEY,
      name TEXT,
      short TEXT,
      description TEXT,
      price NUMERIC
    );
    """)
with open(os.path.join(root, "db_schema.sql"), "w", encoding="utf-8") as f:
